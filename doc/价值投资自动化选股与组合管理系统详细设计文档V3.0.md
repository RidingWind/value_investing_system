#### 2.4.2 参数服务接口

```python
from abc import ABC, abstractmethod

class ParameterService(ABC):
    @abstractmethod
    def register(self, param_def: ParameterDef):
        pass
    
    @abstractmethod
    def get(self, key: str, default: Any = None) -> Any:
        pass
    
    @abstractmethod
    def set(self, key: str, value: Any, operator: str = "system") -> bool:
        pass
    
    @abstractmethod
    def subscribe(self, keys: List[str], callback: callable):
        pass
```

#### 2.4.3 Redis实现类

```python
import json
import redis
import threading
import time
from typing import Dict, List, Any, Callable

class RedisParameterService(ParameterService):
    def __init__(self, redis_client: redis.Redis, namespace: str = "params"):
        self.redis = redis_client
        self.namespace = namespace
        self.local_cache: Dict[str, Any] = {}
        self.definitions: Dict[str, ParameterDef] = {}
        self.subscribers: Dict[str, List[Callable]] = {}
        
        self.pubsub = self.redis.pubsub()
        self.pubsub.subscribe(f"{self.namespace}:events")
        self._start_listener()
    
    def register(self, param_def: ParameterDef):
        self.definitions[param_def.key] = param_def
        stored = self.redis.hget(f"{self.namespace}:values", param_def.key)
        if stored:
            value = json.loads(stored)
        else:
            value = param_def.default_value
            self._persist(param_def.key, value)
        self.local_cache[param_def.key] = value
    
    def get(self, key: str, default: Any = None) -> Any:
        if key in self.local_cache:
            return self.local_cache[key]
        stored = self.redis.hget(f"{self.namespace}:values", key)
        if stored:
            value = json.loads(stored)
            self.local_cache[key] = value
            return value
        return default
    
    def set(self, key: str, value: Any, operator: str = "system") -> bool:
        if key not in self.definitions:
            return False
        param_def = self.definitions[key]
        if not isinstance(value, param_def.value_type):
            return False
        if param_def.validator and not param_def.validator(value):
            return False
        
        self._persist(key, value)
        self._log_change(key, value, operator)
        self.redis.publish(f"{self.namespace}:events", json.dumps({
            "key": key, "value": value, "operator": operator, "timestamp": time.time()
        }))
        return True
    
    def _persist(self, key: str, value: Any):
        self.redis.hset(f"{self.namespace}:values", key, json.dumps(value))
        if self.definitions[key].hot_reloadable:
            self.local_cache[key] = value
    
    def _start_listener(self):
        def listener():
            for message in self.pubsub.listen():
                if message['type'] == 'message':
                    data = json.loads(message['data'])
                    key = data['key']
                    value = data['value']
                    if key in self.definitions and self.definitions[key].hot_reloadable:
                        self.local_cache[key] = value
                    if key in self.subscribers:
                        for cb in self.subscribers[key]:
                            cb(key, value)
        t = threading.Thread(target=listener, daemon=True)
        t.start()
```

```python
    def fetch_daily_quote(self, symbols, date):
```

```python
# core/knowledge/vector_store.py
class VectorStore:
    def __init__(self, param_service: ParameterService):
        import chromadb
        self.client = chromadb.PersistentClient(
            path=param_service.get("knowledge.vector_db_path")
        )
        self.collection = self.client.get_or_create_collection("documents")
        self.embedder = self._get_embedder()
    
    def add_documents(self, docs: List[Document]):
        texts = [doc.content for doc in docs]
        embeddings = self.embedder.encode(texts)
        ids = [doc.id for doc in docs]
        metadatas = [doc.metadata for doc in docs]
        self.collection.add(ids=ids, embeddings=embeddings, metadatas=metadatas, documents=texts)
    
    def similarity_search(self, query: str, k: int = 5) -> List[Document]:
        query_emb = self.embedder.encode([query])[0]
        results = self.collection.query(query_embeddings=[query_emb], n_results=k)
        return self._to_documents(results)

# core/knowledge/hybrid_search.py
class HybridSearchEngine:
    def __init__(self, vector_store: VectorStore, keyword_index: Optional[Elasticsearch] = None):
        self.vector_store = vector_store
        self.keyword_index = keyword_index
    
    def search(self, query: str, top_k: int = 5) -> List[Document]:
        vector_docs = self.vector_store.similarity_search(query, k=top_k*2)
        if self.keyword_index:
            keyword_docs = self.keyword_index.search(query, size=top_k*2)
            return self._reciprocal_rank_fusion(vector_docs, keyword_docs, top_k)
        return vector_docs[:top_k]
```

```python
class PEBasedValuation(ValuationStrategy):
    def __init__(self, param_service: ParameterService, category_key: str):
        self.param_service = param_service
        self.pe_low = self.param_service.get(f"valuation.{category_key}.pe.low")
        self.pe_high = self.param_service.get(f"valuation.{category_key}.pe.high")
        self.overvalued_multiplier = self.param_service.get("valuation.overvalued_multiplier", 1.5)
    
    def valuate(self, stock: Stock) -> ValuationResult:
        if stock.pe_ttm <= self.pe_low:
            status = ValuationStatus.UNDERVALUED
        elif stock.pe_ttm >= self.pe_high * self.overvalued_multiplier:
            status = ValuationStatus.OVERVALUED
        else:
            status = ValuationStatus.FAIR
        return ValuationResult(status, target_price=stock.eps * self.pe_low)

class RNAVValuation(ValuationStrategy):
    def __init__(self, param_service: ParameterService):
        self.param_service = param_service
        self.undervalued_discount = self.param_service.get("valuation.rnav.undervalued_discount", 0.33)
    
    def valuate(self, stock: Stock) -> ValuationResult:
        rnav_per_share = RNAVCalculator(self.param_service).calculate(stock)
        discount = 1 - stock.price / rnav_per_share
        status = ValuationStatus.UNDERVALUED if discount >= self.undervalued_discount else ValuationStatus.FAIR
        return ValuationResult(status, discount, target_price=rnav_per_share * 0.9)

class ValuationEngine:
    def __init__(self, param_service: ParameterService):

            StockCategory.STABLE_GROWTH: PEBasedValuation(param_service, "stable"),
            StockCategory.FAST_GROWTH: PEGBasedValuation(param_service),
            StockCategory.HIDDEN_ASSET: RNAVValuation(param_service),
        }
```

```python
class TechnicalSafetyFilter:
    def __init__(self, param_service: ParameterService):
        self.atr_multiplier = self.param_service.get("timing.atr_multiplier", 2.0)
        self.ma_period = self.param_service.get("timing.ma_period", 20)
    
    def pass_through(self, stock: Stock, portfolio: Portfolio) -> bool:
        if stock.symbol in portfolio.holdings:
            position = portfolio.holdings[stock.symbol]
            atr_tolerance = self.atr_multiplier * stock.atr_14
            if stock.price < stock.ma(self.ma_period) and (position.highest_price - stock.price) > atr_tolerance:
                return False
        return True
```
